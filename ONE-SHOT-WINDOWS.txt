Copy/paste this whole block into PowerShell. Keep ATS-One-Staffing-ERP.zip in Downloads.

$ErrorActionPreference = 'Stop'
$zip = Get-ChildItem "$HOME\Downloads" -Filter "ATS-One-Staffing-ERP*.zip" -File |
       Sort-Object LastWriteTime -Descending |
       Select-Object -First 1
if (-not $zip) { throw "ATS-One-Staffing-ERP ZIP not found in Downloads." }
$dest = Join-Path $HOME 'ATS-One-Local'
if (Test-Path $dest) { Remove-Item $dest -Recurse -Force }
New-Item -ItemType Directory -Path $dest -Force | Out-Null
Expand-Archive -LiteralPath $zip.FullName -DestinationPath $dest -Force
$run = Join-Path $dest 'ATS-One-Staffing-ERP\RUN-WINDOWS.ps1'
if (-not (Test-Path $run)) { throw "RUN-WINDOWS.ps1 missing after extraction." }
powershell.exe -NoProfile -ExecutionPolicy Bypass -File $run
